# --- ڈیٹا بیس کی شروعات ---
def init_db():
    conn = sqlite3.connect("showroom_online.db")
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS customers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT, cnic TEXT, phone TEXT, address TEXT,
            guarantor_name TEXT, guarantor_cnic TEXT, guarantor_phone TEXT, guarantor_address TEXT,
            bike_model TEXT, engine_no TEXT, chassis_no TEXT, color TEXT,
            total_price REAL, down_payment REAL, monthly_installment REAL,
            total_months INTEGER, paid_installments INTEGER DEFAULT 0,
            due_day INTEGER, cust_reminder_phone TEXT, owner_reminder_phone TEXT,
            cust_reminder_day INTEGER, owner_reminder_day INTEGER, entry_date TEXT
        )
    ''')
    conn.commit()
    conn.close()

init_db()

# --- روٹس (Routes) ---
@app.route('/')
def home():
    conn = sqlite3.connect("showroom_online.db")
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM customers")
    rows = cursor.fetchall()
    conn.close()
    return render_template_string(HTML_TEMPLATE, customers=rows)

@app.route('/add', methods=['POST'])
def add_customer():
    name = request.form['name']
    cnic = request.form['cnic']
    phone = request.form['phone']
    address = request.form['address']
    
    g_name = request.form['g_name']
    g_cnic = request.form['g_cnic']
    g_phone = request.form['g_phone']
    g_address = request.form['g_address']
    
    bike_model = request.form['bike_model']
    engine_no = request.form['engine_no']
    chassis_no = request.form['chassis_no']
    color = request.form['color']
    
    price = float(request.form['price'])
    down = float(request.form['down'])
    months = int(request.form['months'])
    due_day = int(request.form['due_day'])
    
    cust_rem_phone = request.form['cust_rem_phone']
    cust_rem_day = int(request.form['cust_rem_day'])
    owner_rem_phone = request.form['owner_rem_phone']
    owner_rem_day = int(request.form['owner_rem_day'])
    
    monthly = (price - down) / months
    entry_date = datetime.now().strftime("%Y-%m-%d")
    
    conn = sqlite3.connect("showroom_online.db")
    cursor = conn.cursor()
    cursor.execute('''
        INSERT INTO customers (
            name, cnic, phone, address,
            guarantor_name, guarantor_cnic, guarantor_phone, guarantor_address,
            bike_model, engine_no, chassis_no, color,
            total_price, down_payment, monthly_installment, total_months, paid_installments,
            due_day, cust_reminder_phone, owner_reminder_phone, cust_reminder_day, owner_reminder_day, entry_date
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 0, ?, ?, ?, ?, ?, ?)
    ''', (name, cnic, phone, address, g_name, g_cnic, g_phone, g_address, bike_model, engine_no, chassis_no, color, price, down, monthly, months, due_day, cust_rem_phone, owner_rem_phone, cust_rem_day, owner_rem_day, entry_date))
    conn.commit()
    conn.close()
    return redirect(url_for('home'))

@app.route('/pay/<int:cust_id>')
def pay_installment(cust_id):
    conn = sqlite3.connect("showroom_online.db")
    cursor = conn.cursor()
    cursor.execute("SELECT paid_installments, total_months FROM customers WHERE id = ?", (cust_id,))
    row = cursor.fetchone()
    if row:
        paid, total = row
        if paid < total:
            cursor.execute("UPDATE customers SET paid_installments = ? WHERE id = ?", (paid + 1, cust_id))
            conn.commit()
        conn.close()
    return redirect(url_for('home'))
